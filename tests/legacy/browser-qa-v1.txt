async function check(name, fn) {
  try { const details = await fn(); report.tests.push({ name, passed: true, ...(details ? { details } : {}) }); console.log('PASS ' + name); }
  catch (error) { report.tests.push({ name, passed: false, error: error.message }); console.log('FAIL ' + name + ': ' + error.message.split('\n')[0]); }
}

async function submit(text, shortcut = false) {
  await page.locator('#input-text').fill(text);
  if (shortcut) await page.locator('#input-text').press('Control+Enter');
  else await page.locator('#transcribe-button').click();
  await page.waitForFunction(() => !document.getElementById('transcribe-button').disabled && !document.getElementById('result-section').hidden);
}

async function actionGeometry() {
  return page.evaluate(() => {
    const box = selector => {
      const { x, y, width, height, right, bottom } = document.querySelector(selector).getBoundingClientRect();
      return { x, y, width, height, right, bottom };
    };
    return { button: box('#transcribe-button'), hint: box('.keyboard-hint'), colors: box('.color-control'), field: box('#input-text') };
  });
}

function assertActionGeometry({ button, hint, colors, field }) {
  assert.ok(hint.x >= button.right, 'Le raccourci est à droite du bouton.');
  assert.ok(Math.abs((hint.y + hint.height / 2) - (button.y + button.height / 2)) < 2, 'Bouton et raccourci sur la même ligne.');
  assert.ok(Math.abs(button.x - field.x) < 2, 'Le bouton de transcription est aligné à gauche.');
  assert.ok(colors.y > button.bottom, 'Les couleurs sont sur la ligne suivante.');
  assert.ok(Math.abs(colors.x - field.x) < 2, 'Les couleurs sont alignées à gauche.');
}

function luminance(color) {
  const channels = color.startsWith('#')
    ? color.slice(1).match(/../g).map(v => parseInt(v, 16))
    : color.match(/[\d.]+/g).slice(0, 3).map(Number);
  return channels.map(c => c / 255).map(c => c <= .04045 ? c / 12.92 : ((c + .055) / 1.055) ** 2.4)
    .reduce((sum, c, i) => sum + c * [.2126, .7152, .0722][i], 0);
}
function contrast(a, b) {
  const first = luminance(a), second = luminance(b);
  return (Math.max(first, second) + .05) / (Math.min(first, second) + .05);
}
